N <- 5000; n <- 30

3. Metropolis-Hastings (MCMC)

              Simulations
            

            
f <- function(x) dgamma(x, shape = 3, rate = 2)
N <- 10000; chain <- numeric(N); chain[1] <- 1

for(i in 2:N) {
  cand <- rnorm(1, mean = chain[i-1], sd = 0.5)
  alpha <- if(cand > 0) min(1, f(cand) / f(chain[i-1])) else 0
  chain[i] <- if(runif(1) < alpha) cand else chain[i-1]
}
chain_clean <- chain[1000:N]
